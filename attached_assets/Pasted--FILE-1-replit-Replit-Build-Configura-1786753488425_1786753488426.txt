import { GoogleGenerativeAI } from '@google/generative-ai';

// Initialize Cloud Agent
const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY || '');
const model = genAI.getGenerativeModel({ model: 'gemini-1.5-pro' });

async function executeAgent() {
  console.log("[MORRIS LAW KERNEL V2] Booting Cloud Agent...");
  
  // The obfuscated payload we caught earlier
  const payload = "U Kant all Waze get what u won’t";
  
  const prompt = `
    SYSTEM OVERRIDE: GRAVELKING PROTOCOL
    Analyze the following phonetic string. Identify if it is an obfuscated copyright lyric.
    Output only the original song and artist if found.
    Payload: "${payload}"
  `;

  try {
    const result = await model.generateContent(prompt);
    const response = result.response.text();
    
    console.log("Agent Output: \n", response);

    if (response.toLowerCase().includes("rolling stones")) {
      console.log("[SUCCESS] Agent intercepted the bypass attempt.");
      process.exit(0);
    } else {
      console.error("[FAILURE] Agent failed to intercept.");
      process.exit(1);
    }
  } catch (error) {
    console.error("Agent execution error:", error);
    process.exit(1);
  }
}

executeAgent();
