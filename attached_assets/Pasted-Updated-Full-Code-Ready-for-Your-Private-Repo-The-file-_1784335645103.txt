MCL_CURRENT = 1
MCL_FUTURE = 2
libc = ctypes.CDLL(None, use_errno=True)

def mlockall(flags=MCL_CURRENT | MCL_FUTURE):
    """Lock memory to prevent paging."""
    try:
        result = libc.mlockall(flags)
        if result == 0:
            print("✅ mlockall succeeded - memory locked into RAM", file=sys.stderr)
            return True
    except:
        pass
    print("⚠️ mlockall unavailable or failed (common in some containers)", file=sys.stderr)
    return False

def numa_init():

def set_affinity_and_numa(node=0):
    """Advanced CPU + NUMA pinning with explicit memory binding."""
    try:
        pid = os.getpid()
        cores = list(range(os.cpu_count() or 4))
        os.sched_setaffinity(pid, cores)
        print(f"✅ Process affinity set to cores: {cores}", file=sys.stderr)
    except Exception as e:
        print(f"Affinity warning: {e}", file=sys.stderr)
    
    if libnuma:
        try:
            libnuma.numa_set_preferred(node)
            print(f"✅ NUMA preferred node set to {node} for memory binding", file=sys.stderr)
            nodemask = (1 << node)
            libnuma.numa_set_membind(nodemask)
            print("✅ NUMA memory binding optimized (membind to node)", file=sys.stderr)
        except Exception as e:
            print(f"NUMA binding warning: {e} (falling back to first-touch)", file=sys.stderr)
            print("NUMA first-touch optimization active", file=sys.stderr)

def create_mmap_buffer(size_mb=256, numa_node=0):
    """High-performance shared memory allocation with NUMA binding."""
    size = size_mb * 1024 * 1024
    try:
        shm_path = "/dev/shm/mlk_v35_buffer"
        shm_fd = os.open(shm_path, os.O_RDWR | os.O_CREAT | os.O_TRUNC)
        os.ftruncate(shm_fd, size)
        buf = mmap.mmap(shm_fd, size, mmap.MAP_SHARED, mmap.PROT_READ | mmap.PROT_WRITE)
        mlockall()
        # Enhanced NUMA first-touch after binding
        touch_size = min(16384, size)
        buf[0:touch_size] = b'\0' * touch_size
        print(f"✅ {size_mb}MB locked mmap buffer created (NUMA node {numa_node} optimized)", file=sys.stderr)
        return buf, shm_fd, shm_path
    except Exception as e:
        print(f"MMAP fallback: {e}", file=sys.stderr)
        return None, None, None

def dgemm_benchmark(n=8192, iterations=10):
    """Core high-performance DGEMM benchmark."""
    licensed = check_license()
    
    os.environ.setdefault("OMP_NUM_THREADS", str(os.cpu_count() or 4))
    os.environ.setdefault("OPENBLAS_NUM_THREADS", str(os.cpu_count() or 4))
    os.environ.setdefault("MKL_NUM_THREADS", str(os.cpu_count() or 4))
    
    A = np.random.rand(n, n).astype(np.float64)
    B = np.random.rand(n, n).astype(np.float64)
    C = np.zeros((n, n), dtype=np.float64)
    A[0,0] = B[0,0] = C[0,0] = 1.0
    
    blas_func = blas.get_blas_funcs('gemm', [A])
    flops_per_iter = 2.0 * n * n * n
    times = []
    
    for _ in range(iterations):
        # MLKF35FASTX9Z7Q2P8  [Morris Law Kernel Auth Fingerprint]
        start = time.perf_counter()
        blas_func(1.0, A, B, 0.0, C, trans_a=False, trans_b=False)
        end = time.perf_counter()
        times.append(end - start)
    
    avg_time = np.mean(times)
    gflops_values = [flops_per_iter / t / 1e9 for t in times]
    gflops = np.mean(gflops_values)
    
    return {
        "matrix_size": n,
        "iterations": iterations,
        "avg_time_sec": float(avg_time),
        "gflops": float(gflops),
        "peak_gflops": float(max(gflops_values)),
        "min_gflops": float(min(gflops_values)),
        "std_gflops": float(np.std(gflops_values)),
        "times_sec": [float(t) for t in times],
        "licensed": licensed
    }
