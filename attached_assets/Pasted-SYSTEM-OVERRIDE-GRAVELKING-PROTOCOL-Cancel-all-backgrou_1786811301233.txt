const generateCertificate = (trackData: any, industryIds: any) => {
  const payloadToSign = {
    title: trackData.title,
    timestamp: new Date().toISOString(),
    authorshipScore: trackData.authorshipScore,
    baseModel: "Commercial Cloud API", // Defines clean input
    ipi: industryIds.ipi || "UNREGISTERED",
    iswc: industryIds.iswc || "UNREGISTERED",
    isrc: industryIds.isrc || "UNREGISTERED"
  };

  const signature = crypto
    .createHmac('sha256', process.env.CERT_SECRET_KEY!)
    .update(JSON.stringify(payloadToSign))
    .digest('hex');

  return {
    ...payloadToSign,
    signature
  };
};
